import os
from datetime import timedelta
from dotenv import load_dotenv

from flask import Flask
from flask_cors import CORS
from flask_sqlalchemy import SQLAlchemy
from flask_login import LoginManager

load_dotenv()

db = SQLAlchemy()
login_manager = LoginManager()


def create_app():
    app = Flask(__name__)

    base_dir = os.path.abspath(os.path.dirname(__file__))
    database_path = os.path.join(base_dir, "wheelq.db")

    app.config["SECRET_KEY"] = os.getenv("SECRET_KEY")
    app.config["ADMIN_EMAIL"] = os.getenv("ADMIN_EMAIL", "")
    app.config["ADMIN_PASSWORD"] = os.getenv("ADMIN_PASSWORD", "")

    if not app.config["SECRET_KEY"]:
        raise RuntimeError("SECRET_KEY must be configured in .env")

    app.config["SQLALCHEMY_DATABASE_URI"] = (
        "sqlite:///" + database_path.replace("\\", "/")
    )

    app.config["SQLALCHEMY_TRACK_MODIFICATIONS"] = False

    app.config["UPLOAD_FOLDER"] = os.path.join(
        base_dir, "uploads", "cars"
    )

    app.config["MAX_CONTENT_LENGTH"] = 5 * 1024 * 1024

    app.config["REMEMBER_COOKIE_HTTPONLY"] = True
    app.config["SESSION_COOKIE_HTTPONLY"] = True
    app.config["SESSION_COOKIE_SAMESITE"] = "Lax"

    app.config["SESSION_COOKIE_SECURE"] = (
        os.getenv("COOKIE_SECURE", "false").lower() == "true"
    )

    app.config["PERMANENT_SESSION_LIFETIME"] = timedelta(
        seconds=int(os.getenv("ADMIN_TOKEN_MAX_AGE", "28800"))
    )

    CORS(
        app,
        supports_credentials=True,
        origins=[
            "http://localhost:5173",
            "http://localhost:5174",
            "http://localhost:5175",
            "http://127.0.0.1:5173",
            "http://127.0.0.1:5174",
            "http://127.0.0.1:5175"
        ],
        allow_headers=["Content-Type", "X-CSRF-Token"]
    )

    db.init_app(app)
    login_manager.init_app(app)

    from backend.models.car import Car
    from backend.models.booking import Booking

    from backend.routes.cars import cars_bp
    from backend.routes.bookings import bookings_bp
    from backend.routes.admin import admin_bp

    app.register_blueprint(cars_bp, url_prefix="/api/cars")
    app.register_blueprint(bookings_bp, url_prefix="/api/bookings")
    app.register_blueprint(admin_bp, url_prefix="/api/admin")

    with app.app_context():
        db.create_all()

    return app

        nullable=True

        db.Date,
        nullable=False

        db.Date,
        nullable=False 
    )

import hmac
import secrets

from flask import Blueprint, jsonify, request, session, current_app
from werkzeug.security import check_password_hash

admin_bp = Blueprint("admin", __name__)


def csrf_valid():
    supplied = request.headers.get("X-CSRF-Token", "")
    expected = session.get("csrf_token", "")

    return bool(expected) and hmac.compare_digest(supplied, expected)


def csrf_required():
    if not csrf_valid():
        return jsonify({"error": "Invalid or expired security token."}), 403
    return None


def admin_required():
    if not session.get("admin_authenticated"):
        return jsonify({"error": "Admin authentication required."}), 401
    return None


@admin_bp.route("/csrf", methods=["GET"])
def get_csrf():
    if not session.get("csrf_token"):
        session["csrf_token"] = secrets.token_urlsafe(32)

    return jsonify({"csrf_token": session["csrf_token"]})


@admin_bp.route("/login", methods=["POST"])
def login():
    csrf_error = csrf_required()
    if csrf_error:
        return csrf_error

    data = request.get_json(silent=True) or {}

    email = str(data.get("email", "")).strip().lower()
    password = str(data.get("password", ""))

    admin_email = current_app.config.get("ADMIN_EMAIL", "").strip().lower()
    password_hash = current_app.config.get("ADMIN_PASSWORD", "")

    if not admin_email or not password_hash:
        return jsonify({"error": "Admin credentials are not configured."}), 500

    email_valid = hmac.compare_digest(email, admin_email)

    try:
        password_valid = check_password_hash(password_hash, password)
    except (ValueError, TypeError):
        password_valid = False

    if not email_valid or not password_valid:
        return jsonify({"error": "Invalid email or password."}), 401

    session.clear()
    session.permanent = True
    session["admin_authenticated"] = True
    session["csrf_token"] = secrets.token_urlsafe(32)

    return jsonify({
        "message": "Login successful.",
        "authenticated": True,
        "csrf_token": session["csrf_token"]
    })


@admin_bp.route("/me", methods=["GET"])
def me():
    authenticated = bool(session.get("admin_authenticated"))

    return jsonify({
        "authenticated": authenticated
    })


@admin_bp.route("/logout", methods=["POST"])
def logout():
    csrf_error = csrf_required()
    if csrf_error:
        return csrf_error

    session.clear()

    return jsonify({"message": "Logged out successfully."})

import secrets
import string
from datetime import datetime

from flask import Blueprint, jsonify, request

from backend import db
from backend.models.booking import Booking
from backend.models.car import Car

bookings_bp = Blueprint("bookings", __name__)


def generate_reference():
    characters = string.ascii_uppercase + string.digits

    while True:
        code = "WQ-" + "".join(
            secrets.choice(characters) for _ in range(6)
        )

        if not Booking.query.filter_by(reference=code).first():
            return code


def booking_data(booking):
    return {
        "id": booking.id,
        "reference": booking.reference,
        "car_id": booking.car_id,
        "car_name": booking.car.name,
        "customer_name": booking.customer_name,
        "pickup_date": booking.pickup_date.isoformat(),
        "return_date": booking.return_date.isoformat(),
        "days": booking.days,
        "daily_rate": booking.daily_rate,
        "total_amount": booking.total_amount,
        "status": booking.status,
        "created_at": booking.created_at.isoformat(),
    }


@bookings_bp.route("", methods=["POST"])
def create_booking():
    data = request.get_json(silent=True) or {}

    required = [
        "car_id",
        "customer_name",
        "phone",
        "pickup_date",
        "return_date",
    ]

    if any(not str(data.get(field, "")).strip() for field in required):
        return jsonify({"error": "Complete all required booking fields."}), 400

    try:
        car_id = int(data["car_id"])
        pickup_date = datetime.strptime(
            data["pickup_date"], "%Y-%m-%d"
        ).date()
        return_date = datetime.strptime(
            data["return_date"], "%Y-%m-%d"
        ).date()
    except (ValueError, TypeError):
        return jsonify({"error": "Invalid vehicle or booking dates."}), 400

    if return_date <= pickup_date:
        return jsonify({
            "error": "Return date must be after pick-up date."
        }), 400

    car = db.session.get(Car, car_id)

    if not car:
        return jsonify({"error": "Vehicle not found."}), 404

    if car.status != "Available":
        return jsonify({
            "error": "This vehicle is currently unavailable."
        }), 409

    overlapping = Booking.query.filter(
        Booking.car_id == car_id,
        Booking.status != "Cancelled",
        Booking.pickup_date < return_date,
        Booking.return_date > pickup_date,
    ).first()

    if overlapping:
        return jsonify({
            "error": "This vehicle is already reserved for those dates."
        }), 409

    customer_name = str(data["customer_name"]).strip()
    phone = str(data["phone"]).strip()
    email = str(data.get("email", "")).strip() or None

    if len(customer_name) > 120 or len(phone) > 30:
        return jsonify({"error": "Customer details are too long."}), 400

    if email and len(email) > 120:
        return jsonify({"error": "Email address is too long."}), 400

    days = (return_date - pickup_date).days
    daily_rate = car.price

    booking = Booking(
        reference=generate_reference(),
        car_id=car.id,
        customer_name=customer_name,
        phone=phone,
        email=email,
        pickup_date=pickup_date,
        return_date=return_date,
        days=days,
        daily_rate=daily_rate,
        total_amount=daily_rate * days,
        status="Pending",
    )

    db.session.add(booking)
    db.session.commit()

    return jsonify(booking_data(booking)), 201


@bookings_bp.route("/<string:reference>", methods=["GET"])
def find_booking(reference):
    phone = request.args.get("phone", "").strip()

    if not phone:
        return jsonify({"error": "Phone number is required."}), 400

    booking = Booking.query.filter_by(
        reference=reference.strip().upper(),
        phone=phone,
    ).first()

    if not booking:
        return jsonify({
            "error": "No booking matches that reference and phone number."
        }), 404

    return jsonify(booking_data(booking))

import os
import uuid
from backend.routes.admin import admin_required, csrf_required
from flask import Blueprint, current_app, jsonify, request, send_from_directory
from backend import db
from backend.models.car import Car

cars_bp = Blueprint("cars", __name__)

ALLOWED_EXTENSIONS = {"png", "jpg", "jpeg", "webp"}


def allowed_file(filename):
    return "." in filename and filename.rsplit(".", 1)[1].lower() in ALLOWED_EXTENSIONS


def car_data(car):
    return {
        "id": car.id,
        "name": car.name,
        "plate": car.plate,
        "category": car.category,
        "price": car.price,
        "status": car.status,
        "image": f"/api/cars/images/{car.image}" if car.image else "",
        "seats": car.seats,
        "transmission": car.transmission,
        "fuel": car.fuel,
    }


@cars_bp.route("", methods=["GET"])
def list_cars():
    cars = Car.query.order_by(Car.id.desc()).all()
    return jsonify([car_data(car) for car in cars])


@cars_bp.route("", methods=["POST"])
def create_car():
    auth_error = admin_required()
    if auth_error:
        return auth_error

    csrf_error = csrf_required()
    if csrf_error:
        return csrf_error
    data = request.form if request.form else (request.get_json(silent=True) or {})

    required = ["name", "plate", "category", "price"]
    if any(not str(data.get(field, "")).strip() for field in required):
        return jsonify({"error": "Complete all required vehicle fields."}), 400

    try:
        price = int(data["price"])
        seats = int(data.get("seats", 5))
    except (ValueError, TypeError):
        return jsonify({"error": "Price and seats must be valid numbers."}), 400

    if price < 0 or seats < 1:
        return jsonify({"error": "Invalid price or seat count."}), 400

    plate = str(data["plate"]).strip().upper()

    if Car.query.filter_by(plate=plate).first():
        return jsonify({"error": "This registration number already exists."}), 409

    image_name = None
    uploaded = request.files.get("image")

    if uploaded and uploaded.filename:
        if not allowed_file(uploaded.filename):
            return jsonify({"error": "Use PNG, JPG, JPEG or WEBP images."}), 400

        extension = uploaded.filename.rsplit(".", 1)[1].lower()
        image_name = f"{uuid.uuid4().hex}.{extension}"
        os.makedirs(current_app.config["UPLOAD_FOLDER"], exist_ok=True)
        uploaded.save(os.path.join(current_app.config["UPLOAD_FOLDER"], image_name))

    car = Car(
        name=str(data["name"]).strip(),
        plate=plate,
        category=str(data["category"]).strip(),
        price=price,
        status=str(data.get("status", "Available")).strip(),
        image=image_name,
        seats=seats,
        transmission=str(data.get("transmission", "Automatic")),
        fuel=str(data.get("fuel", "Petrol")),
    )

    db.session.add(car)
    db.session.commit()
    return jsonify(car_data(car)), 201


@cars_bp.route("/<int:car_id>", methods=["PUT"])
def update_car(car_id):
    auth_error = admin_required()
    if auth_error:
        return auth_error

    csrf_error = csrf_required()
    if csrf_error:
        return csrf_error
    car = db.session.get(Car, car_id)
    if not car:
        return jsonify({"error": "Vehicle not found."}), 404

    data = request.form if request.form else (request.get_json(silent=True) or {})

    if "name" in data:
        car.name = str(data["name"]).strip()

    if "plate" in data:
        plate = str(data["plate"]).strip().upper()
        existing = Car.query.filter_by(plate=plate).first()
        if existing and existing.id != car.id:
            return jsonify({"error": "Registration number already exists."}), 409
        car.plate = plate

    if "category" in data:
        car.category = str(data["category"]).strip()

    if "price" in data:
        try:
            price = int(data["price"])
            if price < 0:
                raise ValueError()
            car.price = price
        except (ValueError, TypeError):
            return jsonify({"error": "Invalid price."}), 400

    if "status" in data:
        if data["status"] not in {"Available", "Booked", "Unavailable"}:
            return jsonify({"error": "Invalid status."}), 400
        car.status = data["status"]

    if "seats" in data:
        try:
            seats = int(data["seats"])
            if seats < 1:
                raise ValueError()
            car.seats = seats
        except (ValueError, TypeError):
            return jsonify({"error": "Invalid seat count."}), 400

    for field in ["transmission", "fuel"]:
        if field in data:
            setattr(car, field, data[field])

    uploaded = request.files.get("image")
    if uploaded and uploaded.filename:
        if not allowed_file(uploaded.filename):
            return jsonify({"error": "Unsupported image format."}), 400

        extension = uploaded.filename.rsplit(".", 1)[1].lower()
        image_name = f"{uuid.uuid4().hex}.{extension}"
        os.makedirs(current_app.config["UPLOAD_FOLDER"], exist_ok=True)
        uploaded.save(os.path.join(current_app.config["UPLOAD_FOLDER"], image_name))
        car.image = image_name

    db.session.commit()
    return jsonify(car_data(car))


@cars_bp.route("/<int:car_id>", methods=["DELETE"])
def delete_car(car_id):
    auth_error = admin_required()
    if auth_error:
        return auth_error
    csrf_error = csrf_required()
    if csrf_error:
        return csrf_error
    car = db.session.get(Car, car_id)
    if not car:
        return jsonify({"error": "Vehicle not found."}), 404

    if car.image:
        image_path = os.path.join(current_app.config["UPLOAD_FOLDER"], car.image)
        if os.path.isfile(image_path):
            os.remove(image_path)

    db.session.delete(car)
    db.session.commit()
    return jsonify({"message": "Vehicle deleted successfully."})


@cars_bp.route("/images/<filename>", methods=["GET"])
def get_car_image(filename):
    return send_from_directory(current_app.config["UPLOAD_FOLDER"], filename)

# Wheel-Q API contract (what the frontend expects)

The frontend calls these three endpoints. Build the Flask routes to match.
All bodies are JSON. Dates are `YYYY-MM-DD`. Money is whole Kenya shillings (KES).
Errors return the right status code and `{ "error": "Plain message the customer can read." }`.

## GET /api/cars?start=2026-11-23&end=2026-11-26

Returns every car in the fleet, each marked free or not for those dates.

```json
[
  {
    "id": 4,
    "name": "Subaru Forester",
    "category": "SUV",
    "seats": 5,
    "transmission": "Automatic",
    "fuel": "Petrol",
    "daily_rate": 7500,
    "plate": "KDC 356K",
    "available": true
  }
]
```

## POST /api/bookings

Request:

```json
{
  "car_id": 4,
  "start": "2026-11-23",
  "end": "2026-11-26",
  "name": "Faith A",
  "phone": "+254712345678",
  "email": "name@example.com",
  "notes": ""
}
```

Response `201`:

```json
{
  "reference": "WQ-4F7K",
  "status": "reserved",
  "car_id": 4,
  "car_name": "Subaru Forester",
  "plate": "KDC 356K",
  "start": "2026-11-23",
  "end": "2026-11-26",
  "days": 3,
  "total": 22500
}
```

Errors: `409` when the car is already booked for any of those dates (the frontend shows
"Someone just reserved this car"), `404` for an unknown car, `400` for bad input.
The backend must re-check availability here. The browser's view can be out of date.

## GET /api/bookings/WQ-4F7K?phone=%2B254712345678

Returns the same shape as the POST response. Only returns the booking when the
reference AND the phone number both match. Otherwise `404`. Never return the phone number.

## Rules the frontend currently assumes (confirm each with the client)

1. Rental days = return date minus pick-up date. 23 Nov to 26 Nov is 3 days.
2. A car returned on a date can be picked up again the same day.
3. Price is a flat daily rate. No weekend rates, discounts or deposit.
4. Reserving does not take payment. How payment works is still to be decided.
5. Phone numbers are Kenyan mobiles (07xx or 01xx), sent as +254XXXXXXXXX.
6. Booking references look like `WQ-` plus 4 letters or digits.

import { API_URL } from './config';
import { ApiError } from './apiError';

let csrfToken = '';

async function request(path, options = {}) {
  let response;

  const isFormData = options.body instanceof FormData;

  try {
    response = await fetch(`${API_URL}${path}`, {
      ...options,
      credentials: 'include',
      headers: {
        ...(isFormData ? {} : { 'Content-Type': 'application/json' }),
        ...(csrfToken ? { 'X-CSRF-Token': csrfToken } : {}),
        ...options.headers,
      },
    });
  } catch {
    throw new ApiError('Could not reach the server.', 0);
  }

  let body = null;

  try {
    body = await response.json();
  } catch {

  }

  if (!response.ok) {
    throw new ApiError(
      body?.error || 'Something went wrong.',
      response.status
    );
  }

  return body;
}

export async function adminGetCsrf() {

  });

  csrfToken = result.csrf_token;

export function findBooking(reference, phone) {
  const query = `phone=${encodeURIComponent(phone)}`;

  return request(
    `/api/bookings/${encodeURIComponent(reference)}?${query}`
  );
}

  return request('/api/cars');
}

function vehiclePayload(payload) {
  if (payload instanceof FormData) {
    return { body: payload };
  }

  const formData = new FormData();

  Object.entries(payload).forEach(([key, value]) => {
    if (key === 'image') {
      if (value instanceof File) {
        formData.append('image', value);
      }
      return;
    }

    if (value !== undefined && value !== null) {
      formData.append(key, String(value));
    }
  });

  return { body: formData };
}

export function adminCreateCar(payload) {
  const { body } = vehiclePayload(payload);

  return request('/api/cars', {
    method: 'POST',
    body,
  });
}

export function adminUpdateCar(id, payload) {
  const { body } = vehiclePayload(payload);

  return request(`/api/cars/${id}`, {
    method: 'PUT',
    body,
  });
}

export function adminDeleteCar(id) {
  return request(`/api/cars/${id}`, {
    method: 'DELETE',
  });
}

    pickup: "",
    return: ""
  });

  const [category, setCategory] = useState("All");
  const [filter, setFilter] = useState("All");
  const [cars, setCars] = useState([]);
  const [loading, setLoading] = useState(true);
  const [fleetError, setFleetError] = useState("");

    try {
      setBookingSaving(true);
      setBookingError("");
      setBookingResult(null);

      const result = await createBooking(payload);
      setBookingResult(result);

    } catch (error) {
      setBookingError(
        error.message || "Could not complete your reservation."
      );
    } finally {
      setBookingSaving(false);
    }

      setLookupError("");

        bookingCode.trim(),

      setLookupError(error.message || "Booking not found.");
    } finally {
      setLookupLoading(false);
    }

    async function loadFleet() {

        setLoading(true);
        setFleetError("");

            ? car.image.startsWith("http")
              ? car.image

        }));

      } catch (error) {
        if (active) {
          setCars([]);
          setFleetError(
            "We couldn't load our vehicles right now. Please try again shortly."
          );
        }
      } finally {
        if (active) setLoading(false);
      }
    }

    loadFleet();

    return () => {
      active = false;
    };
  }, []);
  const availableCars = cars.filter(
    (car) => car.status === "Available"
  );

  const filtered = availableCars.filter(
    (car) => filter === "All" || car.category === filter

      dates.pickup &&
      dates.return &&
      dates.return < dates.pickup
    ) {
      alert("Return date cannot be earlier than pick-up date.");
      return;
    }

    setFilter(category);
    scrollFleet();
  }

  return (
    <>

        onBook={() => document.getElementById("booking")?.scrollIntoView({
          behavior: "smooth"
        })}
        onFind={() => {
          setBookingCode("");
          setBookingPhone("");

          setLookupError("");
          setFindOpen(true);

              setCategory={setCategory}
              onSearch={search}
            />

        </section>

        <ServiceStrip/>

        <Categories
          onSelect={(selected) => {
            setFilter(selected);
            scrollFleet();
          }}
          vehicles={availableCars}
        />

        <Fleet
          vehicles={filtered}
          onBook={setVehicle}
        />

        {loading && (
          <p className="empty-message">
            Loading available vehicles...
          </p>
        )}

        {!loading && fleetError && (
          <p className="empty-message">{fleetError}</p>
        )}

        {!loading && !fleetError && availableCars.length === 0 && (
          <p className="empty-message">
            No vehicles are currently available. Please check back soon.
          </p>
        )}

                type="submit"

                disabled={lookupLoading}

            {foundBooking && (

                </p>
                <p>Status: {foundBooking.status}</p>
              </div>
            )}
          </div>
        </div>
      )}
    </>
  );
}

import { SAMPLE_CARS, sampleBlockedRanges } from './sampleData';
import { daysBetween, rangesOverlap } from './lib/dates';
import { normalizePhone } from './lib/validate';
import { ApiError } from './apiError';
const bookings = [];
const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

function isFree(carId, start, end) {
  const taken = [
    ...(sampleBlockedRanges()[carId] || []),
    ...bookings.filter((b) => b.car_id === carId),
  ];
  return !taken.some((range) => rangesOverlap(start, end, range.start, range.end));
}

function makeReference() {
  const letters = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  let code = '';
  for (let i = 0; i < 4; i += 1) {
    code += letters[Math.floor(Math.random() * letters.length)];
  }
  return `WQ-${code}`;
}

function publicView({ phone, ...booking }) {
  return booking;
}

export async function listCars(start, end) {
  await wait(350);
  return SAMPLE_CARS.map((car) => ({ ...car, available: isFree(car.id, start, end) }));
}

export async function createBooking(payload) {
  await wait(500);
  const car = SAMPLE_CARS.find((c) => c.id === payload.car_id);
  if (!car) {
    throw new ApiError('That car does not exist.', 404);
  }
  if (!isFree(car.id, payload.start, payload.end)) {
    throw new ApiError('That car is no longer free for those dates.', 409);
  }
  const days = daysBetween(payload.start, payload.end);
  const booking = {
    reference: makeReference(),
    status: 'reserved',
    car_id: car.id,
    car_name: car.name,
    plate: car.plate,
    start: payload.start,
    end: payload.end,
    days,
    total: car.daily_rate * days,
    phone: payload.phone,
  };
  bookings.push(booking);
  return publicView(booking);
}

export async function findBooking(reference, phone) {
  await wait(350);
  const match = bookings.find(
    (b) => b.reference === reference.trim().toUpperCase() && b.phone === normalizePhone(phone),
  );
  if (!match) {
    throw new ApiError('No booking matches that reference and phone number.', 404);
  }
  return publicView(match);
}

import { useEffect, useRef, useState } from 'react';
import TextField from './TextField';
import BookingSummary from './BookingSummary';
import BookingConfirmation from './BookingConfirmation';
import { createBooking } from '../api';
import { rentalTotal } from '../lib/money';
import { normalizePhone, validateBooking } from '../lib/validate';

export default function BookingDialog({ car, start, end,  days, onClose, onBooked }) {
  const dialogRef = useRef(null);
  const [form, setForm] = useState({ name: '', phone: '', email: '', notes: '' });
  const [errors, setErrors] = useState({});
  const [submitting, setSubmitting] = useState(false);
  const [serverError, setServerError] = useState('');
  const [result, setResult] = useState(null);

  // <dialog> gives us focus trapping and the Escape key for free once it is opened as a modal.
  useEffect(() => {
    const dialog = dialogRef.current;
    if (dialog && !dialog.open) dialog.showModal();
  }, []);

  function closeDialog() {
    dialogRef.current?.close();
  }

  function handleBackdropClick(event) {
    if (event.target === event.currentTarget) closeDialog();
  }

  function updateField(event) {
    const { name, value } = event.target;
    setForm((current) => ({ ...current, [name]: value }));
  }

  async function handleSubmit(event) {
    event.preventDefault();
    const found = validateBooking(form);
    setErrors(found);
    if (Object.keys(found).length > 0) return;

    setSubmitting(true);
    setServerError('');
    try {
      const { data, sample } = await createBooking({
        car_id: car.id,
        start,
        end,
        name: form.name.trim(),
        phone: normalizePhone(form.phone),
        email: form.email.trim(),
        notes: form.notes.trim(),
      });
      setResult({ booking: data, sample });
      onBooked?.();
    } catch (error) {
      if (error.status === 409) {
        setServerError('Someone just reserved this car for those dates. Close this window and choose another car.');
      } else if (!error.status) {
        setServerError('Couldnâ€™t reach the server. Check your connection and try again.');
      } else {
        setServerError(error.message);
      }
    } finally {
      setSubmitting(false);
    }
  }

  const preview = {
    car_name: car.name,
    plate: car.plate,
    start,
    end,
    days,
    total: rentalTotal(car.daily_rate, days),
  };

  return (
    <dialog
      ref={dialogRef}
      className="dialog"
      aria-labelledby="dialog-title"
      onClose={onClose}
      onClick={handleBackdropClick}
    >
      {result ? (
        <BookingConfirmation booking={result.booking} sample={result.sample} onDone={closeDialog} />
      ) : (
        <form className="dialog__body" onSubmit={handleSubmit} noValidate>
          <h2 className="dialog__title" id="dialog-title">
            Reserve {car.name}
          </h2>

          <BookingSummary details={preview} />

          <TextField
            id="name"
            name="name"
            label="Full name"
            autoComplete="name"
            value={form.name}
            onChange={updateField}
            error={errors.name}
            autoFocus
          />
          <TextField
            id="phone"
            name="phone"
            label="Phone number"
            hint="A Kenyan mobile number, like 0712 345 678."
            type="tel"
            autoComplete="tel"
            value={form.phone}
            onChange={updateField}
            error={errors.phone}
          />
          <TextField
            id="email"
            name="email"
            label="Email address"
            type="email"
            autoComplete="email"
            value={form.email}
            onChange={updateField}
            error={errors.email}
          />
          <TextField
            id="notes"
            name="notes"
            label="Notes (optional)"
            hint="Pick-up time, or anything we should know."
            multiline
            rows={3}
            value={form.notes}
            onChange={updateField}
          />

          {serverError && (
            <p className="notice notice--error" role="alert">
              {serverError}
            </p>
          )}

          <div className="dialog__actions">
            <button type="button" className="btn btn--quiet" onClick={closeDialog}>
              Cancel
            </button>
            <button type="submit" className="btn btn--primary" disabled={submitting}>
              {submitting ? 'Reservingâ€¦' : 'Reserve car'}
            </button>
          </div>
        </form>
      )}
    </dialog>
  );
}

function statusText(fleet, freeCount, days) {
  if (fleet.status === 'loading') return 'Checking which cars are freeâ€¦';
  if (fleet.status === 'error') return '';
  if (freeCount === 0) return 'No cars are free for these dates.';
  const verb = freeCount === 1 ? 'car is' : 'cars are';
  return `${freeCount} ${verb} free for ${days} ${dayWord(days)}.`;
}

const pad = (n) => String(n).padStart(2, '0');

export function toISO(date) {
  return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`;
}

export function fromISO(iso) {
  const [year, month, day] = iso.split('-').map(Number);
  return new Date(year, month - 1, day);
}

export function todayISO() {
  return toISO(new Date());
}

export function addDays(iso, count) {
  const date = fromISO(iso);
  date.setDate(date.getDate() + count);
  return toISO(date);
}

export function daysBetween(startISO, endISO) {
  return Math.round((fromISO(endISO) - fromISO(startISO)) / 86400000);
}

export function rangesOverlap(aStart, aEnd, bStart, bEnd) {
  return aStart < bEnd && bStart < aEnd;
}

export function formatDay(iso) {
  return fromISO(iso).toLocaleDateString('en-GB', {
    weekday: 'short',
    day: 'numeric',
    month: 'short',
  });
}

export function formatShort(iso) {
  return fromISO(iso).toLocaleDateString('en-GB', { day: 'numeric', month: 'short' });
}

export function formatRange(startISO, endISO) {
  return `${formatShort(startISO)} to ${formatShort(endISO)}`;
}

export function dayWord(count) {
  return count === 1 ? 'day' : 'days';
}

export function normalizePhone(input) {
  const cleaned = String(input).replace(/[\s()-]/g, '');
  const match = cleaned.match(/^(?:\+254|254|0)([17]\d{8})$/);
  return match ? `+254${match[1]}` : null;
}

export function validateBooking(form) {
  const errors = {};
  if (form.name.trim().length < 2) {
    errors.name = 'Enter your full name.';
  }
  if (!normalizePhone(form.phone)) {
    errors.phone = 'Enter a Kenyan mobile number, like 0712 345 678.';
  }
  if (!/^\S+@\S+\.\S+$/.test(form.email.trim())) {
    errors.email = 'Enter an email address, like name@example.com.';
  }
  return errors;
}

    setErrors(found);
    if (Object.keys(found).length > 0) return;

    setResult({ status: 'loading', booking: null, sample: false, message: '' });
    try {
      const { data, sample } = await findBooking(form.reference.trim().toUpperCase(), phone);
      setResult({ status: 'found', booking: data, sample, message: '' });
    } catch (error) {
      const message =
        error.status === 404
          ? 'No booking matches that reference and phone number. Check both and try again.'
          : 'Couldnâ€™t reach the server. Check your connection and try again.';
      setResult({ status: 'error', booking: null, sample: false, message });
    }
  }

  return (
    <section className="page">
      <div className="wrap">
        <h1 className="page__title">Find my booking</h1>
        <p className="page__lead">Enter the reference from your confirmation and the phone number you booked with.</p>

        <form className="find-form" onSubmit={handleSubmit} noValidate>
          <TextField
            id="reference"
            name="reference"
            label="Booking reference"
            hint="It looks like WQ-4F7K."
            autoComplete="off"
            value={form.reference}
            onChange={updateField}
            error={errors.reference}
          />
          <TextField
            id="find-phone"
            name="phone"
            label="Phone number"
            type="tel"
            autoComplete="tel"
            value={form.phone}
            onChange={updateField}
            error={errors.phone}
          />
          <div>
            <button type="submit" className="btn btn--primary" disabled={result.status === 'loading'}>
              {result.status === 'loading' ? 'Findingâ€¦' : 'Find booking'}
            </button>
          </div>
        </form>

        <div className="find-result" role="status">
          {result.status === 'error' && <p className="notice notice--error">{result.message}</p>}
          {result.status === 'found' && (
            <>
              <div className="reference">
                <p>Booking reference</p>
                <Plate reg={result.booking.reference} />
              </div>
              <BookingSummary details={result.booking} />
              {result.sample && <p className="notice">This is a sample booking made in this browser tab.</p>}
            </>
          )}
        </div>
      </div>
    </section>
  );
}

import { API_URL, ALLOW_SAMPLE_FALLBACK } from './config';
import { ApiError } from './apiError';
import * as sample from './sampleApi';

async function request(path, options = {}) {
  let response;
  try {
    response = await fetch(`${API_URL}${path}`, {
      headers: { 'Content-Type': 'application/json' },
      ...options,
    });
  } catch {
    throw new ApiError('Could not reach the server.', 0);
  }

  let body = null;
  try {
    body = await response.json();
  } catch {
  }

  if (!response.ok) {
    throw new ApiError(body?.error || 'Something went wrong.', response.status);
  }
  return body;
}

async function withSampleFallback(callServer, callSample) {
  try {
    return { data: await callServer(), sample: false };
  } catch (error) {
    const serverIsDown = error instanceof ApiError && error.status === 0;
    if (ALLOW_SAMPLE_FALLBACK && serverIsDown) {
      return { data: await callSample(), sample: true };
    }
    throw error;
  }
}

export function listCars(start, end) {
  return withSampleFallback(
    () => request(`/api/cars?start=${start}&end=${end}`),
    () => sample.listCars(start, end),
  );
}

export function createBooking(payload) {
  return withSampleFallback(
    () => request('/api/bookings', { method: 'POST', body: JSON.stringify(payload) }),
    () => sample.createBooking(payload),
  );
}

export function findBooking(reference, phone) {
  const query = `phone=${encodeURIComponent(phone)}`;
  return withSampleFallback(
    () => request(`/api/bookings/${encodeURIComponent(reference)}?${query}`),
    () => sample.findBooking(reference, phone),
  );
}

import { useCallback, useEffect, useRef, useState } from 'react';
import Header from './components/Header';
import Footer from './components/Footer';
import BookingDialog from './components/BookingDialog';
import HomePage from './pages/HomePage';
import FindBookingPage from './pages/FindBookingPage';
import { listCars } from './api';
import { addDays, daysBetween, todayISO } from './lib/dates';

export default function App() {
  const [today] = useState(todayISO);
  const [view, setView] = useState('home');
  const [dates, setDates] = useState(() => ({ start: addDays(todayISO(), 1), end: addDays(todayISO(), 4) }));
  const [fleet, setFleet] = useState({ status: 'loading', cars: [], sample: false });
  const [category, setCategory] = useState('All cars');
  const [reserving, setReserving] = useState(null);

  const days = daysBetween(dates.start, dates.end);

  const latestRequest = useRef(0);

  const loadCars = useCallback(
    async ({ quiet = false } = {}) => {
      const requestId = ++latestRequest.current;
      if (!quiet) setFleet((current) => ({ ...current, status: 'loading' }));
      try {
        const { data, sample } = await listCars(dates.start, dates.end);
        if (requestId === latestRequest.current) setFleet({ status: 'ready', cars: data, sample });
      } catch {
        if (requestId === latestRequest.current) setFleet({ status: 'error', cars: [], sample: false });
      }
    },
    [dates.start, dates.end],
  );

  useEffect(() => {
    loadCars();
  }, [loadCars]);

  function changeDates(change) {
    const next = { ...dates, ...change };
    if (next.start < today) next.start = today;
    if (daysBetween(next.start, next.end) < 1) next.end = addDays(next.start, 1);
    setDates(next);
  }

  return (
    <>
      <Header view={view} onNavigate={setView} />

      <main>
        {view === 'home' ? (
          <HomePage
            today={today}
            dates={dates}
            days={days}
            fleet={fleet}
            category={category}
            onCategory={setCategory}
            onDatesChange={changeDates}
            onReserve={setReserving}
            onRetry={() => loadCars()}
          />
        ) : (
          <FindBookingPage />
        )}
      </main>

      <Footer />

      {reserving && (
        <BookingDialog
          key={reserving.id}
          car={reserving}
          start={dates.start}
          end={dates.end}
          days={days}
          onClose={() => setReserving(null)}
          onBooked={() => loadCars({ quiet: true })}
        />
      )}
    </>
  );
}

import { SAMPLE_CARS, sampleBlockedRanges } from './sampleData';
import { daysBetween, rangesOverlap } from './lib/dates';
import { normalizePhone } from './lib/validate';
import { ApiError } from './apiError';

const bookings = [];

const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

function isFree(carId, start, end) {
  const taken = [
    ...(sampleBlockedRanges()[carId] || []),
    ...bookings.filter((b) => b.car_id === carId),
  ];
  return !taken.some((range) => rangesOverlap(start, end, range.start, range.end));
}

function makeReference() {
  const letters = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  let code = '';
  for (let i = 0; i < 4; i += 1) {
    code += letters[Math.floor(Math.random() * letters.length)];
  }
  return `WQ-${code}`;
}

function publicView({ phone, ...booking }) {
  return booking;
}

export async function listCars(start, end) {
  await wait(350);
  return SAMPLE_CARS.map((car) => ({ ...car, available: isFree(car.id, start, end) }));
}

export async function createBooking(payload) {
  await wait(500);
  const car = SAMPLE_CARS.find((c) => c.id === payload.car_id);
  if (!car) {
    throw new ApiError('That car does not exist.', 404);
  }
  if (!isFree(car.id, payload.start, payload.end)) {
    throw new ApiError('That car is no longer free for those dates.', 409);
  }
  const days = daysBetween(payload.start, payload.end);
  const booking = {
    reference: makeReference(),
    status: 'reserved',
    car_id: car.id,
    car_name: car.name,
    plate: car.plate,
    start: payload.start,
    end: payload.end,
    days,
    total: car.daily_rate * days,
    phone: payload.phone,
  };
  bookings.push(booking);
  return publicView(booking);
}

export async function findBooking(reference, phone) {
  await wait(350);
  const match = bookings.find(
    (b) => b.reference === reference.trim().toUpperCase() && b.phone === normalizePhone(phone),
  );
  if (!match) {
    throw new ApiError('No booking matches that reference and phone number.', 404);
  }
  return publicView(match);
}

import { useEffect, useRef, useState } from 'react';
import TextField from './TextField';
import BookingSummary from './BookingSummary';
import BookingConfirmation from './BookingConfirmation';
import { createBooking } from '../api';
import { rentalTotal } from '../lib/money';
import { normalizePhone, validateBooking } from '../lib/validate';

export default function BookingDialog({ car, start, end, days, onClose, onBooked }) {
  const dialogRef = useRef(null);
  const [form, setForm] = useState({ name: '', phone: '', email: '', notes: '' });
  const [errors, setErrors] = useState({});
  const [submitting, setSubmitting] = useState(false);
  const [serverError, setServerError] = useState('');
  const [result, setResult] = useState(null);

  // <dialog> gives us focus trapping and the Escape key for free once it is opened as a modal.
  useEffect(() => {
    const dialog = dialogRef.current;
    if (dialog && !dialog.open) dialog.showModal();
  }, []);

  function closeDialog() {
    dialogRef.current?.close();
  }

  function handleBackdropClick(event) {
    if (event.target === event.currentTarget) closeDialog();
  }

  function updateField(event) {
    const { name, value } = event.target;
    setForm((current) => ({ ...current, [name]: value }));
  }

  async function handleSubmit(event) {
    event.preventDefault();
    const found = validateBooking(form);
    setErrors(found);
    if (Object.keys(found).length > 0) return;

    setSubmitting(true);
    setServerError('');
    try {
      const { data, sample } = await createBooking({
        car_id: car.id,
        start,
        end,
        name: form.name.trim(),
        phone: normalizePhone(form.phone),
        email: form.email.trim(),
        notes: form.notes.trim(),
      });
      setResult({ booking: data, sample });
      onBooked?.();
    } catch (error) {
      if (error.status === 409) {
        setServerError('Someone just reserved this car for those dates. Close this window and choose another car.');
      } else if (!error.status) {
        setServerError('Couldnâ€™t reach the server. Check your connection and try again.');
      } else {
        setServerError(error.message);
      }
    } finally {
      setSubmitting(false);
    }
  }

  const preview = {
    car_name: car.name,
    plate: car.plate,
    start,
    end,
    days,
    total: rentalTotal(car.daily_rate, days),
  };

  return (
    <dialog
      ref={dialogRef}
      className="dialog"
      aria-labelledby="dialog-title"
      onClose={onClose}
      onClick={handleBackdropClick}
    >
      {result ? (
        <BookingConfirmation booking={result.booking} sample={result.sample} onDone={closeDialog} />
      ) : (
        <form className="dialog__body" onSubmit={handleSubmit} noValidate>
          <h2 className="dialog__title" id="dialog-title">
            Reserve {car.name}
          </h2>

          <BookingSummary details={preview} />

          <TextField
            id="name"
            name="name"
            label="Full name"
            autoComplete="name"
            value={form.name}
            onChange={updateField}
            error={errors.name}
            autoFocus
          />
          <TextField
            id="phone"
            name="phone"
            label="Phone number"
            hint="A Kenyan mobile number, like 0712 345 678."
            type="tel"
            autoComplete="tel"
            value={form.phone}
            onChange={updateField}
            error={errors.phone}
          />
          <TextField
            id="email"
            name="email"
            label="Email address"
            type="email"
            autoComplete="email"
            value={form.email}
            onChange={updateField}
            error={errors.email}
          />
          <TextField
            id="notes"
            name="notes"
            label="Notes (optional)"
            hint="Pick-up time, or anything we should know."
            multiline
            rows={3}
            value={form.notes}
            onChange={updateField}
          />

          {serverError && (
            <p className="notice notice--error" role="alert">
              {serverError}
            </p>
          )}

          <div className="dialog__actions">
            <button type="button" className="btn btn--quiet" onClick={closeDialog}>
              Cancel
            </button>
            <button type="submit" className="btn btn--primary" disabled={submitting}>
              {submitting ? 'Reservingâ€¦' : 'Reserve car'}
            </button>
          </div>
        </form>
      )}
    </dialog>
  );
}

const pad = (n) => String(n).padStart(2, '0');

export function toISO(date) {
  return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`;
}

export function fromISO(iso) {
  const [year, month, day] = iso.split('-').map(Number);
  return new Date(year, month - 1, day);
}

export function todayISO() {
  return toISO(new Date());
}

export function addDays(iso, count) {
  const date = fromISO(iso);
  date.setDate(date.getDate() + count);
  return toISO(date);
}

export function daysBetween(startISO, endISO) {
  return Math.round((fromISO(endISO) - fromISO(startISO)) / 86400000);
}

export function rangesOverlap(aStart, aEnd, bStart, bEnd) {
  return aStart < bEnd && bStart < aEnd;
}

export function formatDay(iso) {
  return fromISO(iso).toLocaleDateString('en-GB', {
    weekday: 'short',
    day: 'numeric',
    month: 'short',
  });
}

export function formatShort(iso) {
  return fromISO(iso).toLocaleDateString('en-GB', { day: 'numeric', month: 'short' });
}

export function formatRange(startISO, endISO) {
  return `${formatShort(startISO)} to ${formatShort(endISO)}`;
}

export function dayWord(count) {
  return count === 1 ? 'day' : 'days';
}

export function normalizePhone(input) {
  const cleaned = String(input).replace(/[\s()-]/g, '');
  const match = cleaned.match(/^(?:\+254|254|0)([17]\d{8})$/);
  return match ? `+254${match[1]}` : null;
}

export function validateBooking(form) {
  const errors = {};
  if (form.name.trim().length < 2) {
    errors.name = 'Enter your full name.';
  }
  if (!normalizePhone(form.phone)) {
    errors.phone = 'Enter a Kenyan mobile number, like 0712 345 678.';
  }
  if (!/^\S+@\S+\.\S+$/.test(form.email.trim())) {
    errors.email = 'Enter an email address, like name@example.com.';
  }
  return errors;
}

import { API_URL, ALLOW_SAMPLE_FALLBACK } from './config';
import { ApiError } from './apiError';
import * as sample from './sampleApi';

async function request(path, options = {}) {
  let response;
  try {
    response = await fetch(`${API_URL}${path}`, {
      headers: { 'Content-Type': 'application/json' },
      ...options,
    });
  } catch {
    throw new ApiError('Could not reach the server.', 0);
  }

  let body = null;
  try {
    body = await response.json();
  } catch {
    
  }

  if (!response.ok) {
    throw new ApiError(body?.error || 'Something went wrong.', response.status);
  }
  return body;
}
async function withSampleFallback(callServer, callSample) {
  try {
    return { data: await callServer(), sample: false };
  } catch (error) {
    const serverIsDown = error instanceof ApiError && error.status === 0;
    if (ALLOW_SAMPLE_FALLBACK && serverIsDown) {
      return { data: await callSample(), sample: true };
    }
    throw error;
  }
}

export function listCars(start, end) {
  return withSampleFallback(
    () => request(`/api/cars?start=${start}&end=${end}`),
    () => sample.listCars(start, end),
  );
}

export function createBooking(payload) {
  return withSampleFallback(
    () => request('/api/bookings', { method: 'POST', body: JSON.stringify(payload) }),
    () => sample.createBooking(payload),
  );
}

export function findBooking(reference, phone) {
  const query = `phone=${encodeURIComponent(phone)}`;
  return withSampleFallback(
    () => request(`/api/bookings/${encodeURIComponent(reference)}?${query}`),
    () => sample.findBooking(reference, phone),
  );
}

import { useCallback, useEffect, useRef, useState } from 'react';
import Header from './components/Header';
import Footer from './components/Footer';
import BookingDialog from './components/BookingDialog';
import HomePage from './pages/HomePage';
import FindBookingPage from './pages/FindBookingPage';
import { listCars } from './api';
import { addDays, daysBetween, todayISO } from './lib/dates';

export default function App() {
  const [today] = useState(todayISO);
  const [view, setView] = useState('home');
  const [dates, setDates] = useState(() => ({ start: addDays(todayISO(), 1), end: addDays(todayISO(), 4) }));
  const [fleet, setFleet] = useState({ status: 'loading', cars: [], sample: false });
  const [category, setCategory] = useState('All cars');
  const [reserving, setReserving] = useState(null);

  const days = daysBetween(dates.start, dates.end);

  const latestRequest = useRef(0);

  const loadCars = useCallback(
    async ({ quiet = false } = {}) => {
      const requestId = ++latestRequest.current;
      if (!quiet) setFleet((current) => ({ ...current, status: 'loading' }));
      try {
        const { data, sample } = await listCars(dates.start, dates.end);
        if (requestId === latestRequest.current) setFleet({ status: 'ready', cars: data, sample });
      } catch {
        if (requestId === latestRequest.current) setFleet({ status: 'error', cars: [], sample: false });
      }
    },
    [dates.start, dates.end],
  );

  useEffect(() => {
    loadCars();
  }, [loadCars]);

  function changeDates(change) {
    const next = { ...dates, ...change };
    if (next.start < today) next.start = today;
    if (daysBetween(next.start, next.end) < 1) next.end = addDays(next.start, 1);
    setDates(next);
  }

  return (
    <>
      <Header view={view} onNavigate={setView} />

      <main>
        {view === 'home' ? (
          <HomePage
            today={today}
            dates={dates}
            days={days}
            fleet={fleet}
            category={category}
            onCategory={setCategory}
            onDatesChange={changeDates}
            onReserve={setReserving}
            onRetry={() => loadCars()}
          />
        ) : (
          <FindBookingPage />
        )}
      </main>

      <Footer />

      {reserving && (
        <BookingDialog
          key={reserving.id}
          car={reserving}
          start={dates.start}
          end={dates.end}
          days={days}
          onClose={() => setReserving(null)}
          onBooked={() => loadCars({ quiet: true })}
        />
      )}
    </>
  );
}

import { SAMPLE_CARS, sampleBlockedRanges } from './sampleData';
import { daysBetween, rangesOverlap } from './lib/dates';
import { normalizePhone } from './lib/validate';
import { ApiError } from './apiError';

const bookings = [];

const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

function isFree(carId, start, end) {
  const taken = [
    ...(sampleBlockedRanges()[carId] || []),
    ...bookings.filter((b) => b.car_id === carId),
  ];
  return !taken.some((range) => rangesOverlap(start, end, range.start, range.end));
}

function makeReference() {
  const letters = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  let code = '';
  for (let i = 0; i < 4; i += 1) {
    code += letters[Math.floor(Math.random() * letters.length)];
  }
  return `WQ-${code}`;
}

function publicView({ phone, ...booking }) {
  return booking;
}

export async function listCars(start, end) {
  await wait(350);
  return SAMPLE_CARS.map((car) => ({ ...car, available: isFree(car.id, start, end) }));
}

export async function createBooking(payload) {
  await wait(500);
  const car = SAMPLE_CARS.find((c) => c.id === payload.car_id);
  if (!car) {
    throw new ApiError('That car does not exist.', 404);
  }
  if (!isFree(car.id, payload.start, payload.end)) {
    throw new ApiError('That car is no longer free for those dates.', 409);
  }
  const days = daysBetween(payload.start, payload.end);
  const booking = {
    reference: makeReference(),
    status: 'reserved',
    car_id: car.id,
    car_name: car.name,
    plate: car.plate,
    start: payload.start,
    end: payload.end,
    days,
    total: car.daily_rate * days,
    phone: payload.phone,
  };
  bookings.push(booking);
  return publicView(booking);
}

export async function findBooking(reference, phone) {
  await wait(350);
  const match = bookings.find(
    (b) => b.reference === reference.trim().toUpperCase() && b.phone === normalizePhone(phone),
  );
  if (!match) {
    throw new ApiError('No booking matches that reference and phone number.', 404);
  }
  return publicView(match);
}

import { useEffect, useRef, useState } from 'react';
import TextField from './TextField';
import BookingSummary from './BookingSummary';
import BookingConfirmation from './BookingConfirmation';
import { createBooking } from '../api';
import { rentalTotal } from '../lib/money';
import { normalizePhone, validateBooking } from '../lib/validate';

export default function BookingDialog({ car, start, end,  days, onClose, onBooked }) {
  const dialogRef = useRef(null);
  const [form, setForm] = useState({ name: '', phone: '', email: '', notes: '' });
  const [errors, setErrors] = useState({});
  const [submitting, setSubmitting] = useState(false);
  const [serverError, setServerError] = useState('');
  const [result, setResult] = useState(null);

  useEffect(() => {
    const dialog = dialogRef.current;
    if (dialog && !dialog.open) dialog.showModal();
  }, []);

  function closeDialog() {
    dialogRef.current?.close();
  }

  function handleBackdropClick(event) {
    if (event.target === event.currentTarget) closeDialog();
  }

  function updateField(event) {
    const { name, value } = event.target;
    setForm((current) => ({ ...current, [name]: value }));
  }

  async function handleSubmit(event) {
    event.preventDefault();
    const found = validateBooking(form);
    setErrors(found);
    if (Object.keys(found).length > 0) return;

    setSubmitting(true);
    setServerError('');
    try {
      const { data, sample } = await createBooking({
        car_id: car.id,
        start,
        end,
        name: form.name.trim(),
        phone: normalizePhone(form.phone),
        email: form.email.trim(),
        notes: form.notes.trim(),
      });
      setResult({ booking: data, sample });
      onBooked?.();
    } catch (error) {
      if (error.status === 409) {
        setServerError('Someone just reserved this car for those dates. Close this window and choose another car.');
      } else if (!error.status) {
        setServerError('Couldnâ€™t reach the server. Check your connection and try again.');
      } else {
        setServerError(error.message);
      }
    } finally {
      setSubmitting(false);
    }
  }

  const preview = {
    car_name: car.name,
    plate: car.plate,
    start,
    end,
    days,
    total: rentalTotal(car.daily_rate, days),
  };

  return (
    <dialog
      ref={dialogRef}
      className="dialog"
      aria-labelledby="dialog-title"
      onClose={onClose}
      onClick={handleBackdropClick}
    >
      {result ? (
        <BookingConfirmation booking={result.booking} sample={result.sample} onDone={closeDialog} />
      ) : (
        <form className="dialog__body" onSubmit={handleSubmit} noValidate>
          <h2 className="dialog__title" id="dialog-title">
            Reserve {car.name}
          </h2>

          <BookingSummary details={preview} />

          <TextField
            id="name"
            name="name"
            label="Full name"
            autoComplete="name"
            value={form.name}
            onChange={updateField}
            error={errors.name}
            autoFocus
          />
          <TextField
            id="phone"
            name="phone"
            label="Phone number"
            hint="A Kenyan mobile number, like 0712 345 678."
            type="tel"
            autoComplete="tel"
            value={form.phone}
            onChange={updateField}
            error={errors.phone}
          />
          <TextField
            id="email"
            name="email"
            label="Email address"
            type="email"
            autoComplete="email"
            value={form.email}
            onChange={updateField}
            error={errors.email}
          />
          <TextField
            id="notes"
            name="notes"
            label="Notes (optional)"
            hint="Pick-up time, or anything we should know."
            multiline
            rows={3}
            value={form.notes}
            onChange={updateField}
          />

          {serverError && (
            <p className="notice notice--error" role="alert">
              {serverError}
            </p>
          )}

          <div className="dialog__actions">
            <button type="button" className="btn btn--quiet" onClick={closeDialog}>

            </button>

  if (freeCount === 0) return 'No cars are free for these dates.';
  const verb = freeCount === 1 ? 'car is' : 'cars are';
  return `${freeCount} ${verb} free for ${days} ${dayWord(days)}.`;
}

const pad = (n) => String(n).padStart(2, '0');

export function toISO(date) {
  return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`;
}

export function fromISO(iso) {
  const [year, month, day] = iso.split('-').map(Number);
  return new Date(year, month - 1, day);
}

export function todayISO() {
  return toISO(new Date());
}

export function addDays(iso, count) {
  const date = fromISO(iso);
  date.setDate(date.getDate() + count);
  return toISO(date);
}

export function daysBetween(startISO, endISO) {
  return Math.round((fromISO(endISO) - fromISO(startISO)) / 86400000);
}

export function rangesOverlap(aStart, aEnd, bStart, bEnd) {
  return aStart < bEnd && bStart < aEnd;
}

export function formatDay(iso) {
  return fromISO(iso).toLocaleDateString('en-GB', {
    weekday: 'short',
    day: 'numeric',
    month: 'short',
  });
}

export function formatShort(iso) {
  return fromISO(iso).toLocaleDateString('en-GB', { day: 'numeric', month: 'short' });
}

export function formatRange(startISO, endISO) {
  return `${formatShort(startISO)} to ${formatShort(endISO)}`;
}

export function dayWord(count) {
  return count === 1 ? 'day' : 'days';
}

export function normalizePhone(input) {
  const cleaned = String(input).replace(/[\s()-]/g, '');
  const match = cleaned.match(/^(?:\+254|254|0)([17]\d{8})$/);
  return match ? `+254${match[1]}` : null;
}

export function validateBooking(form) {
  const errors = {};
  if (form.name.trim().length < 2) {
    errors.name = 'Enter your full name.';
  }
  if (!normalizePhone(form.phone)) {
    errors.phone = 'Enter a Kenyan mobile number, like 0712 345 678.';
  }
  if (!/^\S+@\S+\.\S+$/.test(form.email.trim())) {
    errors.email = 'Enter an email address, like name@example.com.';
  }
  return errors;
}

import { useState } from 'react';
import TextField from '../components/TextField';
import BookingSummary from '../components/BookingSummary';
import Plate from '../components/Plate';
import { findBooking } from '../api';
import { normalizePhone } from '../lib/validate';

export default function FindBookingPage() {
  const [form, setForm] = useState({ reference: '', phone: '' });
  const [errors, setErrors] = useState({});
  const [result, setResult] = useState({ status: 'idle', booking: null, sample: false, message: '' });

  function updateField(event) {
    const { name, value } = event.target;
    setForm((current) => ({ ...current, [name]: value }));
  }

  async function handleSubmit(event) {
    event.preventDefault();
    const found = {};
    const phone = normalizePhone(form.phone);
    if (!form.reference.trim()) found.reference = 'Enter your booking reference.';
    if (!phone) found.phone = 'Enter the phone number you booked with, like 0712 345 678.';
    setErrors(found);
    if (Object.keys(found).length > 0) return;

    setResult({ status: 'loading', booking: null, sample: false, message: '' });
    try {
      const { data, sample } = await findBooking(form.reference.trim().toUpperCase(), phone);
      setResult({ status: 'found', booking: data, sample, message: '' });
    } catch (error) {
      const message =
        error.status === 404
          ? 'No booking matches that reference and phone number. Check both and try again.'
          : 'Couldnâ€™t reach the server. Check your connection and try again.';
      setResult({ status: 'error', booking: null, sample: false, message });
    }
  }

  return (
    <section className="page">
      <div className="wrap">
        <h1 className="page__title">Find my booking</h1>
        <p className="page__lead">Enter the reference from your confirmation and the phone number you booked with.</p>

        <form className="find-form" onSubmit={handleSubmit} noValidate>
          <TextField
            id="reference"
            name="reference"
            label="Booking reference"
            hint="It looks like WQ-4F7K."
            autoComplete="off"
            value={form.reference}
            onChange={updateField}
            error={errors.reference}
          />
          <TextField
            id="find-phone"
            name="phone"
            label="Phone number"
            type="tel"
            autoComplete="tel"
            value={form.phone}
            onChange={updateField}
            error={errors.phone}
          />
          <div>
            <button type="submit" className="btn btn--primary" disabled={result.status === 'loading'}>
              {result.status === 'loading' ? 'Findingâ€¦' : 'Find booking'}
            </button>
          </div>
        </form>

        <div className="find-result" role="status">
          {result.status === 'error' && <p className="notice notice--error">{result.message}</p>}
          {result.status === 'found' && (
            <>
              <div className="reference">
                <p>Booking reference</p>
                <Plate reg={result.booking.reference} />
              </div>
              <BookingSummary details={result.booking} />
              {result.sample && <p className="notice">This is a sample booking made in this browser tab.</p>}
            </>
          )}
        </div>
      </div>
    </section>
  );
}
